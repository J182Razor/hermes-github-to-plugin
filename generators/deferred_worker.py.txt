def sdk_major(installed):
    if not isinstance(installed, str) or not re.fullmatch(r'(?:0|[1-9][0-9]*)\.(?:0|[1-9][0-9]*)\.(?:0|[1-9][0-9]*)', installed, re.ASCII):
        raise RuntimeError('unsupported MCP version')
    major = installed.split('.')[0]
    if major not in ('1', '2'):
        raise RuntimeError('unsupported MCP version')
    return major

def tool_result(result, major):
    if major not in ('1', '2'):
        raise RuntimeError('unsupported MCP version')
    flag = getattr(result, 'isError' if major == '1' else 'is_error', None)
    alternate = 'is_error' if major == '1' else 'isError'
    if hasattr(result, alternate) and getattr(result, alternate) is not flag:
        raise RuntimeError('unsupported MCP tool result shape')
    serialized = result.model_dump(mode='json', by_alias=True, exclude_none=True)
    if type(flag) is not bool or not isinstance(serialized, dict) or serialized.get('isError') is not flag:
        raise RuntimeError('unsupported MCP tool result shape')
    if 'is_error' in serialized and serialized['is_error'] is not flag:
        raise RuntimeError('unsupported MCP tool result shape')
    return {'success': not flag, 'result': serialized}

def unique_pairs(pairs):
    result = {}
    for key, value in pairs:
        if key in result: raise ValueError('duplicate JSON key')
        result[key] = value
    return result
